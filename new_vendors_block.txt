      const raw = await apiClient.get<any>(`${VENDORS_PREFIX}/products/summary`);
      return raw?.data || raw;

      console.warn("[VendorsService] getProductSummary API failed, using mock:", err);
      await delay(150);
      const products = mockVendorProducts;
      return {
        totalProducts:   products.length,
        activeProducts:  products.filter((p) => p.status === "active").length,
        outOfStockCount: products.filter((p) => p.status === "out_of_stock").length,
        inactiveCount:   products.filter(
          (p) => p.status === "inactive" || p.status === "discontinued"
        ).length,

                (products.reduce((s, p) => s + p.margin, 0) / products.length) * 10
